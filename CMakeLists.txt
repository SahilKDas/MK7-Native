cmake_minimum_required(VERSION 3.25)
project(MK7Native VERSION 0.2.0 LANGUAGES CXX)

option(MK7_NATIVE_BUILD_TESTS "Build the MK7-Native test suite" ON)
option(MK7_NATIVE_BUILD_RECOMPILER "Build the ARMv6K CTR static recompiler" ON)
option(MK7_NATIVE_STATIC_RECOMPILATION "Emit and compile verbose per-instruction C++ (high disk usage)" OFF)
set(MK7_ROM_PATH "" CACHE FILEPATH "Path to the legally obtained MK7 CIA (read in place; never copied)")
set(CTRTOOL_PATH "" CACHE FILEPATH "Path to ctrtool")
set(MK7_SYMBOL_MAP "" CACHE FILEPATH "Optional mk7re/Ghidra/IDA function map")
set(MK7_COVERAGE_PERMILLE 798 CACHE STRING "Minimum validated executable-code closure in permille")
set(MK7_INDIRECT_ENTRY_SEEDS "0x005EA0D4;0x0053CE54;0x0053CEBC" CACHE STRING "Semicolon-separated ARM function entries observed through indirect branches")
set(MK7_TEXT_SIZE 0x004EB57C CACHE STRING "Authoritative executable text size from the supported CIA exheader")
set(MK7_GENERATED_SHARDS 64 CACHE STRING "Generated C++ translation units for scalable compilation")
set(MK7_ROM_SHA512 "8b10af4bf0347d4ed6076d4b29f36e6d2ab76f6ef60459958ac72842b7345877fb4721992c04664149edce7c294715803d4261b93a9553ef6e85616b71ce0225" CACHE STRING "Expected CIA SHA-512")
set(MK7_NATIVE_GENERATED_ENTRY "" CACHE FILEPATH "Pre-generated entry C++ source (advanced override)")
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/SDL/CMakeLists.txt")
  message(FATAL_ERROR "SDL is missing; run git submodule update --init --recursive")
endif()
add_subdirectory(external/SDL EXCLUDE_FROM_ALL)
find_package(Vulkan REQUIRED)

add_library(mk7_host src/host/application.cpp src/host/input.cpp src/host/layout.cpp src/recomp/pica200.cpp src/recomp/pica_raster.cpp)
target_compile_features(mk7_host PUBLIC cxx_std_23)
target_include_directories(mk7_host PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(mk7_host PUBLIC SDL3::SDL3-static Vulkan::Vulkan)

add_library(mk7_formats src/formats/format.cpp)
add_library(MK7Native::formats ALIAS mk7_formats)
target_compile_features(mk7_formats PUBLIC cxx_std_23)
target_include_directories(mk7_formats PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
if(MSVC)
  target_compile_options(mk7_formats PUBLIC /std:c++latest PRIVATE /W4 /permissive-)
else()
  target_compile_options(mk7_formats PUBLIC -std=c++2c PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(mk7-inspect apps/mk7-inspect/main.cpp)
target_compile_features(mk7-inspect PRIVATE cxx_std_23)
target_link_libraries(mk7-inspect PRIVATE MK7Native::formats)

if(MK7_NATIVE_BUILD_RECOMPILER)
  set(ARM_RECOMP_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/arm-recomp-core")
  if(NOT EXISTS "${ARM_RECOMP_CORE_DIR}/common/arm_decode.cpp")
    message(FATAL_ERROR "arm-recomp-core is missing; run git submodule update --init")
  endif()
  set(ARM_RECOMP_SOURCES "${ARM_RECOMP_CORE_DIR}/common/arm_decode.cpp" "${ARM_RECOMP_CORE_DIR}/common/thumb_decode.cpp" "${ARM_RECOMP_CORE_DIR}/common/arm_ir.cpp" "${ARM_RECOMP_CORE_DIR}/common/interpreter.cpp" tools/recompiler/armv6k_ctr/arm_codegen.cpp tools/recompiler/armv6k_ctr/armv6k_decode.cpp)
  add_executable(mk7-recompile tools/recompiler/main.cpp ${ARM_RECOMP_SOURCES})
  target_compile_features(mk7-recompile PRIVATE cxx_std_23)
  target_include_directories(mk7-recompile PRIVATE "${ARM_RECOMP_CORE_DIR}/common" tools/recompiler/armv6k_ctr)
endif()

set(MK7_ENTRY_SOURCE src/recomp/entry_stub.cpp)
set(MK7_ENTRY_SOURCES "${MK7_ENTRY_SOURCE}")
if(MK7_ROM_PATH AND MK7_NATIVE_STATIC_RECOMPILATION)
  if(NOT MK7_NATIVE_BUILD_RECOMPILER)
    message(FATAL_ERROR "MK7_ROM_PATH requires MK7_NATIVE_BUILD_RECOMPILER=ON")
  endif()
  if(NOT EXISTS "${MK7_ROM_PATH}")
    message(FATAL_ERROR "MK7_ROM_PATH does not exist: ${MK7_ROM_PATH}")
  endif()
  if(NOT CTRTOOL_PATH)
    find_program(CTRTOOL_PATH ctrtool REQUIRED)
  elseif(NOT EXISTS "${CTRTOOL_PATH}")
    message(FATAL_ERROR "CTRTOOL_PATH does not exist: ${CTRTOOL_PATH}")
  endif()
  if(MK7_SYMBOL_MAP AND NOT EXISTS "${MK7_SYMBOL_MAP}")
    message(FATAL_ERROR "MK7_SYMBOL_MAP does not exist: ${MK7_SYMBOL_MAP}")
  endif()
  set(MK7_GEN_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
  set(MK7_CODE_BIN "${MK7_GEN_DIR}/exefs/code.bin")
  set(MK7_ENTRY_SOURCE "${MK7_GEN_DIR}/mk7_entry.cpp")
  set(MK7_BYTE_MAP "${MK7_GEN_DIR}/mk7_entry.map.csv")
  set(MK7_ENTRY_SOURCES "${MK7_ENTRY_SOURCE}")
  math(EXPR _last_shard "${MK7_GENERATED_SHARDS} - 1")
  foreach(_shard RANGE 0 ${_last_shard})
    list(APPEND MK7_ENTRY_SOURCES "${MK7_GEN_DIR}/mk7_entry.part${_shard}.cpp")
  endforeach()
  set(_symbol_args)
  set(_seed_args)
  foreach(_seed IN LISTS MK7_INDIRECT_ENTRY_SEEDS)
    list(APPEND _seed_args --seed "${_seed}")
  endforeach()
  set(_symbol_dep)
  if(MK7_SYMBOL_MAP)
    list(APPEND _symbol_args --symbols "${MK7_SYMBOL_MAP}")
    list(APPEND _symbol_dep "${MK7_SYMBOL_MAP}")
  endif()
  add_custom_command(OUTPUT ${MK7_ENTRY_SOURCES} "${MK7_BYTE_MAP}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${MK7_GEN_DIR}/exefs"
    COMMAND ${CMAKE_COMMAND} "-DINPUT=${MK7_ROM_PATH}" "-DEXPECTED=${MK7_ROM_SHA512}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/VerifyRom.cmake"
    COMMAND "${CTRTOOL_PATH}" --quiet --decompresscode "--exefsdir=${MK7_GEN_DIR}/exefs" "${MK7_ROM_PATH}"
    COMMAND $<TARGET_FILE:mk7-recompile> --input "${MK7_CODE_BIN}" --output "${MK7_ENTRY_SOURCE}" --map-output "${MK7_BYTE_MAP}" --entry 0x00100000 --executable-size "${MK7_TEXT_SIZE}" --coverage-permille "${MK7_COVERAGE_PERMILLE}" --shards "${MK7_GENERATED_SHARDS}" ${_symbol_args} ${_seed_args}
    DEPENDS mk7-recompile "${MK7_ROM_PATH}" ${_symbol_dep} cmake/VerifyRom.cmake
    COMMENT "Verify external CIA, extract ExeFS, recompile ARMv6K functions, and audit coverage" VERBATIM)
  add_custom_target(mk7-clean-generated
    COMMAND ${CMAKE_COMMAND}
      "-DBUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}"
      "-DGENERATED_DIR=${MK7_GEN_DIR}"
      "-DOBJECT_DIR=${CMAKE_CURRENT_BINARY_DIR}/CMakeFiles/mk7-run.dir/generated"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CleanGenerated.cmake"
    COMMENT "Remove disposable generated MK7 sources and objects; preserve mk7-run")
  set_source_files_properties(${MK7_ENTRY_SOURCES} PROPERTIES GENERATED TRUE)
  if(MSVC)
    set_property(SOURCE ${MK7_ENTRY_SOURCES} APPEND PROPERTY COMPILE_OPTIONS /Od)
  else()
    set_property(SOURCE ${MK7_ENTRY_SOURCES} APPEND PROPERTY COMPILE_OPTIONS -O0)
  endif()
elseif(MK7_NATIVE_GENERATED_ENTRY)
  if(NOT EXISTS "${MK7_NATIVE_GENERATED_ENTRY}")
    message(FATAL_ERROR "MK7_NATIVE_GENERATED_ENTRY does not exist")
  endif()
  set(MK7_ENTRY_SOURCE "${MK7_NATIVE_GENERATED_ENTRY}")
  set(MK7_ENTRY_SOURCES "${MK7_ENTRY_SOURCE}")
endif()

add_executable(mk7-run apps/mk7-run/main.cpp src/mii/bridge.cpp src/recomp/runtime.cpp ${MK7_ENTRY_SOURCES})
if(NOT MK7_NATIVE_STATIC_RECOMPILATION)
  target_sources(mk7-run PRIVATE src/recomp/compact_executor.cpp
    external/arm-recomp-core/common/arm_decode.cpp
    external/arm-recomp-core/common/thumb_decode.cpp
    external/arm-recomp-core/common/arm_ir.cpp

    tools/recompiler/armv6k_ctr/armv6k_decode.cpp)

  target_compile_definitions(mk7-run PRIVATE MK7_COMPACT_EXECUTION=1)
endif()

target_include_directories(mk7-run PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

  target_link_libraries(mk7-run PRIVATE bcrypt)

  message(FATAL_ERROR "mk7-run currently supports Windows only")
endif()
